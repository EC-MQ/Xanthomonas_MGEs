##genome annotation with Bakta##
for file in genomes_fasta_files/*.fasta; do \
	name=$(basename "$file" .fasta); \
	bakta -d /patho_to_bakta_database/db \
	--keep-contig-headers \
	--output bakta \
	--force $file; done


##MOB search with HMMER##
#mob profiles were downloaded from here https://github.com/macsy-models/CONJScan
#input files are the .faa files produced by Bakta

for file in bakta/*.faa; do \
	name=$(basename $file .faa); \
	hmmscan --cpu 16 -T 33 \
		--tblout mob_search/"$name"mob_search.results.txt \
		-o mob_search/"$name"mob_search.results_extended.txt \
		/path_to_database/conjscan_hmm_profiles/MOB.hmm "$file"; \
done
